## 一、模块概述

### 1.1 功能范围

场景与角色模块负责处理游戏中的场景数据和角色数据，为游戏运行时提供必要的场景信息、地图数据、交互点配置以及角色信息、关系网络等。

### 1.2 与其他模块的关系

```
┌─────────────────────────────────────────────────────────────┐
│                      游戏运行时                              │
│   game_service.py 调用场景/角色数据驱动AI剧情生成           │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                   场景与角色模块                             │
│  • scenes API: 场景详情、地图数据、交互点、执行交互         │
│  • characters API: 角色详情、关系网络                       │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                      数据层                                  │
│  Scene表: scene_key, map_data, interactable_points等        │
│  Character表: character_key, stats, relationships等         │
└─────────────────────────────────────────────────────────────┘
```

### 4.1 scene_service.py

```python
# app/services/scene_service.py

from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from app.models.models import Scene, GameSession
from app.core.exceptions import NotFoundError, ValidationError


class SceneService:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_scene(self, scene_id: int) -> Scene:
        """获取场景详情"""
        scene = await self.db.get(Scene, scene_id)
        if not scene:
            raise NotFoundError("场景")
        return scene

    async def get_scene_by_key(self, script_id: int, scene_key: str) -> Scene:
        """通过scene_key获取场景"""
        stmt = select(Scene).where(
            Scene.script_id == script_id,
            Scene.scene_key == scene_key
        )
        result = await self.db.execute(stmt)
        scene = result.scalar_one_or_none()
        if not scene:
            raise NotFoundError("场景")
        return scene

    async def get_map_data(self, scene_id: int) -> dict:
        """获取场景地图数据"""
        scene = await self.get_scene(scene_id)
        return scene.map_data

    async def get_interactables(self, scene_id: int) -> list[dict]:
        """获取场景交互点列表"""
        scene = await self.get_scene(scene_id)
        return scene.map_data.get("interactable_points", [])

    async def execute_interaction(
        self,
        scene_id: int,
        session_id: int,
        interactable_id: str,
        action_id: str
    ) -> dict:
        """执行交互动作"""
        scene = await self.get_scene(scene_id)
        game_session = await self.db.get(GameSession, session_id)

        if not game_session:
            raise NotFoundError("游戏会话")

        # 查找交互点
        interactables = scene.map_data.get("interactable_points", [])
        interactable = next(
            (i for i in interactables if i["id"] == interactable_id),
            None
        )
        if not interactable:
            raise NotFoundError("交互点")

        # 检查动作是否可用
        available_actions = interactable.get("actions", [])
        if action_id not in available_actions:
            raise ValidationError(f"该交互点不支持动作: {action_id}")

        # 检查前置条件
        conditions = interactable.get("conditions", {}).get(action_id, [])
        for cond in conditions:
            if not self._check_condition(cond, game_session):
                raise ValidationError("不满足交互条件")

        # 计算交互效果
        effects = self._calculate_effects(interactable, action_id)

        return {
            "success": True,
            "message": self._get_interaction_message(interactable, action_id),
            "effects": effects,
            "ai_trigger": {
                "should_generate": True,
                "context": f"玩家对{interactable['name']}执行了{action_id}"
            }
        }

    def _check_condition(self, condition: dict, session: GameSession) -> bool:
        """检查交互条件"""
        cond_type = condition.get("type")

        if cond_type == "flag":
            flag = condition.get("flag")
            expected = condition.get("value", True)
            return session.story_flags.get(flag) == expected

        if cond_type == "stat":
            stat = condition.get("stat")
            operator = condition.get("operator", ">=")
            value = condition.get("value", 0)
            current = session.player_stats.get(stat, 0)
            return self._compare(current, operator, value)

        return True

    def _compare(self, a: int, op: str, b: int) -> bool:
        """比较操作"""
        ops = {
            "==": lambda x, y: x == y,
            "!=": lambda x, y: x != y,
            ">": lambda x, y: x > y,
            "<": lambda x, y: x < y,
            ">=": lambda x, y: x >= y,
            "<=": lambda x, y: x <= y,
        }
        return ops.get(op, lambda x, y: True)(a, b)

    def _calculate_effects(self, interactable: dict, action_id: str) -> list[dict]:
        """计算交互效果"""
        effects = []

        # 标记已交互
        effects.append({
            "type": "set_flag",
            "target": f"{interactable['id']}_{action_id}",
            "value": True
        })

        # 特定动作的效果
        if interactable["type"] == "exit" and action_id == "enter":
            effects.append({
                "type": "change_scene",
                "target": interactable.get("target_scene"),
                "position": interactable.get("target_position")
            })

        return effects

    def _get_interaction_message(self, interactable: dict, action_id: str) -> str:
        """获取交互提示消息"""
        action_messages = {
            "examine": f"你仔细观察{interactable['name']}...",
            "talk": f"你走向{interactable['name']}，准备交谈...",
            "touch": f"你伸手触碰{interactable['name']}...",
            "enter": f"你走向{interactable['name']}...",
        }
        return action_messages.get(action_id, f"你对{interactable['name']}执行了{action_id}")

# app/services/character_service.py

from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from sqlalchemy.orm import selectinload
from app.models.models import Character, Script
from app.core.exceptions import NotFoundError


class CharacterService:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_character(self, character_id: int) -> Character:
        """获取角色详情"""
        character = await self.db.get(Character, character_id)
        if not character:
            raise NotFoundError("角色")
        return character

    async def get_character_by_key(self, script_id: int, character_key: str) -> Character:
        """通过character_key获取角色"""
        stmt = select(Character).where(
            Character.script_id == script_id,
            Character.character_key == character_key
        )
        result = await self.db.execute(stmt)
        character = result.scalar_one_or_none()
        if not character:
            raise NotFoundError("角色")
        return character

    async def get_relationships(self, character_id: int) -> dict:
        """获取角色关系网络"""
        character = await self.get_character(character_id)
        relationships_data = character.relationships or {}

        # 获取关联角色的基本信息
        relationships = []
        for rel in relationships_data.get("relations", []):
            target_key = rel.get("target_key")
            if target_key:
                # 查询目标角色
                stmt = select(Character).where(
                    Character.script_id == character.script_id,
                    Character.character_key == target_key
                )
                result = await self.db.execute(stmt)
                target = result.scalar_one_or_none()

                if target:
                    relationships.append({
                        "target_id": target.id,
                        "target_name": target.name,
                        "target_key": target.character_key,
                        "relation_type": rel.get("type", "neutral"),
                        "description": rel.get("description", ""),
                        "attitude": rel.get("attitude", "neutral"),
                        "intensity": rel.get("intensity", 50)
                    })

        return {
            "character_id": character.id,
            "character_name": character.name,
            "relationships": relationships
        }

    async def list_script_characters(self, script_id: int) -> list[Character]:
        """获取剧本的所有角色"""
        stmt = (
            select(Character)
            .where(Character.script_id == script_id)
            .order_by(Character.sort_order)
        )
        result = await self.db.execute(stmt)
        return list(result.scalars().all())


character_service = CharacterService

# app/api/v1/scenes.py

from fastapi import APIRouter, Depends
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.deps import get_db, get_current_user
from app.services.scene_service import scene_service
from app.schemas.scene import (
    SceneOut, MapDataOut, InteractablesOut,
    InteractRequest, InteractResponse
)

router = APIRouter(prefix="/scenes", tags=["scenes"])


@router.get("/{scene_id}", response_model=SceneOut)
async def get_scene(
    scene_id: int,
    db: AsyncSession = Depends(get_db),
    user = Depends(get_current_user)
):
    """获取场景详情"""
    service = scene_service(db)
    scene = await service.get_scene(scene_id)
    return {"code": 200, "data": scene}


@router.get("/{scene_id}/map", response_model=MapDataOut)
async def get_scene_map(
    scene_id: int,
    db: AsyncSession = Depends(get_db),
    user = Depends(get_current_user)
):
    """获取场景地图数据"""
    service = scene_service(db)
    map_data = await service.get_map_data(scene_id)
    return {"code": 200, "data": map_data}


@router.get("/{scene_id}/interactables", response_model=InteractablesOut)
async def get_interactables(
    scene_id: int,
    db: AsyncSession = Depends(get_db),
    user = Depends(get_current_user)
):
    """获取场景交互点列表"""
    service = scene_service(db)
    interactables = await service.get_interactables(scene_id)
    return {"code": 200, "data": {"interactables": interactables}}


@router.post("/{scene_id}/interact", response_model=InteractResponse)
async def execute_interaction(
    scene_id: int,
    request: InteractRequest,
    db: AsyncSession = Depends(get_db),
    user = Depends(get_current_user)
):
    """执行交互动作"""
    service = scene_service(db)
    result = await service.execute_interaction(
        scene_id,
        request.session_id,
        request.interactable_id,
        request.action_id
    )
    return {"code": 200, "data": result}
```

from pydantic import BaseModel
from typing import Optional, List, Dict, Any


class SceneData(BaseModel):
    id: int
    script_id: int
    scene_key: str
    name: str
    description: Optional[str] = None
    background_image: Optional[str] = None
    background_music: Optional[str] = None
    connected_scenes: Optional[List[Dict[str, Any]]] = None
    on_enter: Optional[Dict[str, Any]] = None
    on_exit: Optional[Dict[str, Any]] = None

    class Config:
        from_attributes = True


class MapData(BaseModel):
    width: int
    height: int
    grid_size: int
    layers: List[Dict[str, Any]]
    walkable_areas: List[Dict[str, Any]]
    obstacles: List[Dict[str, Any]]
    spawn_points: List[Dict[str, Any]]
    lighting: Optional[Dict[str, Any]] = None


class Interactable(BaseModel):
    id: str
    name: str
    type: str
    position: List[int]
    icon: Optional[str] = None
    description: Optional[str] = None
    actions: List[str] = []
    conditions: Optional[Dict[str, List[Dict[str, Any]]]] = None
    npc_id: Optional[str] = None
    target_scene: Optional[str] = None
    target_position: Optional[List[int]] = None


class InteractRequest(BaseModel):
    session_id: int
    interactable_id: str
    action_id: str


class InteractEffect(BaseModel):
    type: str
    target: str
    value: Optional[Any] = None
    position: Optional[List[int]] = None


class AITrigger(BaseModel):
    should_generate: bool
    context: str


class InteractResult(BaseModel):
    success: bool
    message: str
    effects: List[InteractEffect]
    ai_trigger: AITrigger


# Response schemas
class SceneOut(BaseModel):
    code: int = 200
    data: SceneData


class MapDataOut(BaseModel):
    code: int = 200
    data: MapData


class InteractablesOut(BaseModel):
    code: int = 200
    data: Dict[str, List[Interactable]]


class InteractResponse(BaseModel):
    code: int = 200
    data: InteractResult

import api from "./api";

export const sceneService = {

  interact: (sceneId: number, data: { session_id: number; interactable_id: string; action_id: string }) =>
