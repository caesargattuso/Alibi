## 一、模块概述

### 1.1 功能范围

剧本互动功能模块负责处理用户与剧本之间的互动行为，包括：
- 剧本收藏/取消收藏
- 剧本评分与评论
- 用户密码修改
- 游戏历史记录查看
- 对话历史记录查看

### 1.2 与其他模块的关系

```
┌─────────────────────────────────────────────────────────────┐
│                      用户模块                                │
│  密码修改、游戏历史、成就                                    │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                   剧本互动模块                               │
│  • 收藏/取消收藏                                             │
│  • 评分/评论                                                 │
│  • 游戏历史列表                                              │
│  • 对话历史查询                                              │
└─────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────┐
│                      数据层                                  │
│  ScriptFavorite表: user_id + script_id                      │
│  ScriptRating表: user_id + script_id + rating + comment     │
│  GameSession表: 游戏历史                                    │
│  DialogLog表: 对话记录                                      │
└─────────────────────────────────────────────────────────────┘
```

## 二、API设计

### 2.1 剧本收藏

#### 收藏剧本

```
POST /api/v1/scripts/{script_id}/favorite
```

响应：
```json
{
  "code": 200,
  "data": {
    "favorited": true,
    "favorite_count": 42
  }
}
```

#### 取消收藏

```
DELETE /api/v1/scripts/{script_id}/favorite
```

响应：
```json
{
  "code": 200,
  "data": {
    "favorited": false,
    "favorite_count": 41
  }
}
```

#### 检查收藏状态

```
GET /api/v1/scripts/{script_id}/favorite
```

响应：
```json
{
  "code": 200,
  "data": {
    "favorited": true
  }
}
```

### 2.2 剧本评分

#### 评分/更新评分

```
POST /api/v1/scripts/{script_id}/rate
```

请求：
```json
{
  "rating": 5,
  "comment": "非常精彩的剧本！"
}
```

响应：
```json
{
  "code": 200,
  "data": {
    "rating": 5,
    "comment": "非常精彩的剧本！",
    "average_rating": 4.5,
    "rating_count": 128
  }
}
```


查询参数：

| 参数 | 类型 | 说明 |
|------|------|------|
| page | int | 页码，默认1 |
| per_page | int | 每页数量，默认20 |

响应：
```json
{
  "code": 200,
  "data": {
    "items": [
      {
        "id": 1,
        "user_id": 1,
        "username": "player1",
        "avatar_url": null,
        "rating": 5,
        "comment": "非常精彩！",
        "created_at": "2026-05-30T12:00:00Z"
      }
    ],
    "total": 128,
    "page": 1,
    "per_page": 20
  }
}
```

## 三、服务层设计

### 3.1 favorite_service.py

```python
class FavoriteService:
    async def toggle_favorite(self, user_id: int, script_id: int) -> dict:
        """切换收藏状态（已收藏则取消，未收藏则添加）"""

    async def check_favorite(self, user_id: int, script_id: int) -> bool:
        """检查是否已收藏"""

    async def get_favorite_count(self, script_id: int) -> int:
        """获取剧本收藏数"""

    async def list_user_favorites(self, user_id: int, page: int, per_page: int) -> dict:
        """获取用户收藏列表"""
```

### 3.2 rating_service.py

```python
class RatingService:
    async def rate_script(self, user_id: int, script_id: int, rating: int, comment: str | None) -> dict:
        """评分/更新评分，同时更新剧本平均分"""

    async def get_script_ratings(self, script_id: int, page: int, per_page: int) -> dict:
        """获取剧本评分列表"""

    async def _update_script_average(self, script_id: int):
        """更新剧本平均评分（内部方法）"""
```

## 四、Pydantic Schema

### 4.1 新增Schema

```python
# favorite
class FavoriteOut(BaseModel):
    favorited: bool
    favorite_count: int

# rating
class RateRequest(BaseModel):
    rating: int = Field(ge=1, le=5)
    comment: str | None = Field(None, max_length=500)

class RateOut(BaseModel):
    rating: int
    comment: str | None
    average_rating: float
    rating_count: int

class RatingItem(BaseModel):
    id: int
    user_id: int
    username: str
    avatar_url: str | None
    rating: int
    comment: str | None
    created_at: datetime

# password
class PasswordChange(BaseModel):
    old_password: str = Field(min_length=6)
    new_password: str = Field(min_length=6)

# game history
class GameHistoryItem(BaseModel):
    id: int
    script_id: int
    script_title: str
    script_cover: str | None
    player_name: str | None
    status: str
    current_ending: str | None
    game_phase: str | None
    game_time: str | None
    created_at: datetime
    completed_at: datetime | None

# dialog history
class DialogItem(BaseModel):
    id: int
    type: str
    content: str
    metadata: dict | None
    player_input: str | None
    created_at: datetime
```

## 五、前端服务层

### 5.1 扩展 scripts.ts

```typescript
// 新增方法
favorite: (scriptId: number) =>
  api.post<ApiResponse<{ favorited: boolean; favorite_count: number }>>(`/scripts/${scriptId}/favorite`),

unfavorite: (scriptId: number) =>
  api.delete<ApiResponse<{ favorited: boolean; favorite_count: number }>>(`/scripts/${scriptId}/favorite`),

checkFavorite: (scriptId: number) =>
  api.get<ApiResponse<{ favorited: boolean }>>(`/scripts/${scriptId}/favorite`),

rate: (scriptId: number, data: { rating: number; comment?: string }) =>
  api.post<ApiResponse<RateResult>>(`/scripts/${scriptId}/rate`, data),

getRatings: (scriptId: number, params?: { page?: number; per_page?: number }) =>
  api.get<ApiResponse<PaginatedData<RatingItem>>>(`/scripts/${scriptId}/ratings`, { params }),
```
